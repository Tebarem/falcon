cmake_minimum_required(VERSION 3.25)

project(falcon_engine LANGUAGES C CXX)

# Windows (D3D12, Vulkan), Linux (Vulkan) and macOS on Apple silicon (Metal).
if(APPLE)
  enable_language(OBJC OBJCXX)
elseif(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Falcon builds on Windows, Linux and macOS")
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(FALCON_TRACY "Build with the Tracy profiler client (on-demand)" OFF)
set(FALCON_SLANG_VERSION 2026.10.2)
if(WIN32)
  set(fx_slang_default "$ENV{USERPROFILE}/slang-${FALCON_SLANG_VERSION}-windows-x86_64")
elseif(APPLE)
  set(fx_slang_default "$ENV{HOME}/slang-${FALCON_SLANG_VERSION}-macos-aarch64")
else()
  if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(fx_slang_default "$ENV{HOME}/slang-${FALCON_SLANG_VERSION}-linux-aarch64")
  else()
    set(fx_slang_default "$ENV{HOME}/slang-${FALCON_SLANG_VERSION}-linux-x86_64")
  endif()
endif()
set(FALCON_SLANG_ROOT "${fx_slang_default}" CACHE PATH "Slang binary release root (contains bin/, include/, cmake/)")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

include(FalconCompiler)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
include(FalconSDL)
include(FalconDeps)
include(FalconShaders)
include(FalconUi)
include(FalconFsr)
include(FalconDlss)
include(FalconSteamAudio)
include(FalconMsQuic)
include(FalconSteamworks)

enable_testing()

add_subdirectory(tools/shaderc)
add_subdirectory(tools/cook)
add_subdirectory(tools/imgdiff)
add_subdirectory(tools/wavdiff)
add_subdirectory(tools/schemagen)
add_subdirectory(engine)
add_subdirectory(tools/nettest)
add_subdirectory(tools/leveltest)
add_subdirectory(tools/bvhtest)
# (the scripts first: the tests name their files)
add_subdirectory(managed)
add_subdirectory(runtime)
